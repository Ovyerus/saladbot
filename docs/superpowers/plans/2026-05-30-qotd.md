# QOTD (Question of the Day) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a per-guild Question of the Day feature that posts a plain-text question to a configured channel at a configured time daily, marking each question as used and self-disabling when the pool is exhausted.

**Architecture:** Per-guild Oban scheduled jobs (one per guild) mirror the existing CheeseTouch pattern. Business logic lives in `Salad.Qotd`, DB operations in `Salad.Repo.QotdQuestion`, and the Oban worker in `Salad.Scheduled.Qotd`. A single `/qotd` slash command with subcommands handles all admin operations.

**Tech Stack:** Elixir, Ecto/PostgreSQL, Oban (background jobs), Nostrum (Discord), existing `Salad.CommandSystem` conventions.

> **Note:** This codebase has no existing test suite. TDD steps are omitted — implement and verify manually via the bot itself or `iex -S mix`.

defmodule Salad.Repo.QotdQuestion do
  @moduledoc false
  use Ecto.Schema

  import Ecto.Query
  alias Salad.Repo
  alias Ecto.Changeset

  @type id() :: pos_integer()
  @type guild_id() :: pos_integer()
  @type content() :: String.t()

  @typep repo_result() :: {:ok, t()} | {:error, Ecto.Changeset.t()}

  @type t() :: %__MODULE__{
          id: id(),
          guild_id: guild_id(),
          content: content(),
          posted_at: NaiveDateTime.t() | nil
        }

  @required ~w(guild_id content)a
  @optional ~w(posted_at)a
  @all @required ++ @optional

  schema "qotd_questions" do
    field :content, :string
    field :posted_at, :naive_datetime

    belongs_to :guild, Repo.Guild, type: :integer
    timestamps()
  end

  def changeset(question, params \\ %{}) do
    question
    |> Changeset.cast(params, @all)
    |> Changeset.validate_required(@required)
  end

  @spec create(guild_id(), content()) :: repo_result()
  def create(guild_id, content) when is_integer(guild_id) and is_binary(content) do
    %__MODULE__{}
    |> changeset(%{guild_id: guild_id, content: content})
    |> Repo.insert()
  end

  @spec delete(t()) :: repo_result()
  def delete(%__MODULE__{} = question) do
    Repo.delete(question)
  end

  @spec get(id()) :: t() | nil
  def get(id) when is_integer(id) do
    Repo.get(__MODULE__, id)
  end

  @spec get_for_guild(guild_id()) :: list(t())
  def get_for_guild(guild_id) when is_integer(guild_id) do
    __MODULE__
    |> where(guild_id: ^guild_id)
    |> order_by(asc: :inserted_at)
    |> Repo.all()
  end

  @spec get_next_unposted(guild_id()) :: t() | nil
  def get_next_unposted(guild_id) when is_integer(guild_id) do
    __MODULE__
    |> where(guild_id: ^guild_id)
    |> where([q], is_nil(q.posted_at))
    |> order_by(asc: :inserted_at)
    |> limit(1)
    |> Repo.one()
  end

  @spec mark_posted(t()) :: repo_result()
  def mark_posted(%__MODULE__{} = question) do
    question
    |> changeset(%{posted_at: NaiveDateTime.utc_now() |> NaiveDateTime.truncate(:second)})
    |> Repo.update()
  end

  @spec search_for_guild(guild_id(), String.t()) :: list(t())
  def search_for_guild(guild_id, text) when is_integer(guild_id) and is_binary(text) do
    pattern = "%#{String.replace(text, "%", "\\%")}%"

    __MODULE__
    |> where(guild_id: ^guild_id)
    |> where([q], ilike(q.content, ^pattern))
    |> order_by(asc: :inserted_at)
    |> limit(25)
    |> Repo.all()
  end
end

# lib/salad/qotd.ex
defmodule Salad.Qotd do
  require Logger
  alias Salad.Repo

  @doc """
  Posts the next unposted question for a guild to its configured QOTD channel.
  Marks the question as posted and reschedules the next job.

  If no questions remain, disables QOTD for the guild and does not reschedule.
  """
  @spec run(pos_integer()) ::
          {:ok, pos_integer(), Repo.QotdQuestion.t()}
          | {:cancel, :no_questions_remaining | :qotd_not_enabled | :guild_not_found}
  def run(guild_id) do
    case Repo.Guild.get(guild_id) do
      nil ->
        Logger.warning("QOTD run for unknown guild #{guild_id}")
        {:cancel, :guild_not_found}

      %Repo.Guild{qotd_channel: nil} ->
        {:cancel, :qotd_not_enabled}

      %Repo.Guild{qotd_channel: channel_id} = guild ->
        run_for_guild(guild_id, channel_id, guild)
    end
  end

  defp run_for_guild(guild_id, channel_id, guild) do
    case Repo.QotdQuestion.get_next_unposted(guild_id) do
      nil ->
        Logger.warning("No QOTD questions remaining for guild #{guild_id}, disabling.")
        Repo.Guild.update(guild, %{qotd_channel: nil, qotd_time: nil})
        {:cancel, :no_questions_remaining}

      %Repo.QotdQuestion{} = question ->
        with {:ok, _} <- Nostrum.Api.create_message(channel_id, question.content),
             {:ok, _} <- Repo.QotdQuestion.mark_posted(question),
             {:ok, _} <- schedule(guild_id) do
          Logger.info("Posted QOTD for guild #{guild_id}, question id=#{question.id}")
          {:ok, channel_id, question}
        else
          err ->
            Logger.error("Failed to run QOTD for guild #{guild_id}: #{inspect(err)}")
            err
        end
    end
  end

  @doc """
  Cancels any existing scheduled QOTD job for the guild and schedules the next one
  based on the guild's configured `qotd_time` (UTC).
  """
  @spec schedule(pos_integer(), Repo.Guild.t() | nil) ::
          {:ok, Oban.Job.t()} | {:error, Ecto.Changeset.t()}
  def schedule(guild_id, guild \\ nil) do
    import Ecto.Query

    # Cancel any existing scheduled job for this guild.
    Oban.Job
    |> where(queue: "qotd")
    |> where(state: "scheduled")
    |> where(fragment("args->>'guild_id' = ?", ^to_string(guild_id)))
    |> Oban.cancel_all_jobs()

    guild = guild || Repo.Guild.get(guild_id)
    run_at = next_run_at(guild.qotd_time)

    %{guild_id: guild_id}
    |> Salad.Scheduled.Qotd.new(scheduled_at: run_at)
    |> Oban.insert()
  end

  @doc """
  Computes the next UTC `DateTime` for a given `Time`.
  If the time is still in the future today (UTC), returns today at that time.
  Otherwise returns tomorrow at that time.
  """
  @spec next_run_at(Time.t()) :: DateTime.t()
  def next_run_at(%Time{} = qotd_time) do
    now = DateTime.utc_now()
    today = DateTime.to_date(now)
    today_run = DateTime.new!(today, qotd_time, "Etc/UTC")

    if DateTime.compare(today_run, now) == :gt do
      today_run
    else
      DateTime.add(today_run, 1, :day)
    end
  end
end

      {:cancel, _reason} -> :ok

# lib/salad/commands/qotd.ex
defmodule Salad.Commands.Qotd do
  @moduledoc false
  import Bitwise
  require Logger
  use Salad.CommandSystem.Command

  alias Nostrum.Api
  alias Salad.Repo

  @impl true
  def description, do: "Question of the Day commands"

  @impl true
  def predicates,
    do: [
      guild_only(),
      permissions([:manage_guild]),
      guild_setup()
    ]

  @impl true
  def options,
    do: [
      %Option{
        name: "enable",
        type: Option.Type.subcommand(),
        description: "Enable QOTD for this server.",
        options: [
          %Option{
            name: "channel",
            type: Option.Type.channel(),
            description: "The channel to post questions in.",
            required: true
          },
          %Option{
            name: "time",
            type: Option.Type.string(),
            description: "When to post each day — paste a Discord timestamp (type @time 9pm tomorrow to get one).",
            required: true
          }
        ]
      },
      %Option{
        name: "disable",
        type: Option.Type.subcommand(),
        description: "Disable QOTD for this server."
      },
      %Option{
        name: "add",
        type: Option.Type.subcommand(),
        description: "Add a question to the pool.",
        options: [
          %Option{
            name: "question",
            type: Option.Type.string(),
            description: "The question to add.",
            required: true
          }
        ]
      },
      %Option{
        name: "remove",
        type: Option.Type.subcommand(),
        description: "Remove a question from the pool.",
        options: [
          %Option{
            name: "question",
            type: Option.Type.string(),
            description: "The question to remove.",
            required: true,
            autocomplete: true
          }
        ]
      },
      %Option{
        name: "list",
        type: Option.Type.subcommand(),
        description: "List all questions in the pool."
      }
    ]

  @impl true
  def autocomplete(%{data: %{options: options}, guild_id: guild_id}) do
    opt = Enum.find(options, fn opt -> opt.focused end)

    case opt && opt.name do
      "question" ->
        results =
          if opt.value && opt.value != "" do
            Repo.QotdQuestion.search_for_guild(guild_id, opt.value)
          else
            Repo.QotdQuestion.get_for_guild(guild_id) |> Enum.take(25)
          end

        Enum.map(results, fn q ->
          label = if String.length(q.content) > 97, do: String.slice(q.content, 0, 97) <> "...", else: q.content
          %{name: label, value: to_string(q.id)}
        end)

      _ ->
        []
    end
  end

  @impl true
  def run(ctx) do
    case ctx.options do
      %{"enable" => _} -> run_enable(ctx)
      %{"disable" => _} -> run_disable(ctx)
      %{"add" => _} -> run_add(ctx)
      %{"remove" => _} -> run_remove(ctx)
      %{"list" => _} -> run_list(ctx)
    end
  end

  # Discord timestamp format: <t:UNIX> or <t:UNIX:format>
  @timestamp_regex ~r/<t:(\d+)(?::[tTdDfFR])?>/

  defp parse_discord_time(str) do
    case Regex.run(@timestamp_regex, str) do
      [_, unix_str] ->
        unix = String.to_integer(unix_str)
        dt = DateTime.from_unix!(unix)
        {:ok, DateTime.to_time(dt)}

      nil ->
        case Integer.parse(str) do
          {unix, ""} ->
            dt = DateTime.from_unix!(unix)
            {:ok, DateTime.to_time(dt)}

          _ ->
            :error
        end
    end
  end

  defp run_enable(ctx) do
    %{
      "channel" => %{value: channel},
      "time" => %{value: time_str}
    } = ctx.options["enable"].value

    with true <- channel.type == Nostrum.Constants.ChannelType.guild_text() or :not_text_channel,
         {:ok, qotd_time} <- parse_discord_time(time_str),
         %Repo.Guild{} = guild <- Repo.Guild.get(ctx.guild_id),
         {:ok, _updated} <-
           Repo.Guild.update(guild, %{qotd_channel: channel.id, qotd_time: qotd_time}) do
      Salad.Qotd.schedule(ctx.guild_id, %{guild | qotd_time: qotd_time})

      reply(ctx, %{
        type: 4,
        data: %{
          content:
            "QOTD has been enabled in <##{channel.id}>. Questions will be posted daily at #{Time.to_string(qotd_time)} UTC."
        }
      })
    else
      :not_text_channel ->
        reply(ctx, %{
          type: 4,
          data: %{
            content: "The channel you selected is not a plain text channel.",
            flags: 1 <<< 6
          }
        })

      :error ->
        reply(ctx, %{
          type: 4,
          data: %{
            content:
              "Please provide a time using a Discord timestamp — type `@time 9pm tomorrow` in any channel to get one, then paste it here.",
            flags: 1 <<< 6
          }
        })

      nil ->
        Logger.error("/qotd enable reached nil guild for #{ctx.guild_id}")

        reply(ctx, %{
          type: 4,
          data: %{
            content: "Something went wrong fetching this server. Try `/setup` first.",
            flags: 1 <<< 6
          }
        })

      err ->
        Logger.error("Failed to enable QOTD for guild #{ctx.guild_id}: #{inspect(err)}")

        reply(ctx, %{
          type: 4,
          data: %{content: "An error occurred.", flags: 1 <<< 6}
        })
    end
  end

  defp run_disable(ctx) do
    import Ecto.Query

    with %Repo.Guild{} = guild <- Repo.Guild.get(ctx.guild_id),
         {:ok, _updated} <-
           Repo.Guild.update(guild, %{qotd_channel: nil, qotd_time: nil}) do
      # Cancel any pending Oban job for this guild.
      Oban.Job
      |> where(queue: "qotd")
      |> where(state: "scheduled")
      |> where(fragment("args->>'guild_id' = ?", ^to_string(ctx.guild_id)))
      |> Oban.cancel_all_jobs()

      reply(ctx, %{
        type: 4,
        data: %{content: "QOTD has been disabled.", flags: 1 <<< 6}
      })
    else
      err ->
        Logger.error("Failed to disable QOTD for guild #{ctx.guild_id}: #{inspect(err)}")
        reply(ctx, %{type: 4, data: %{content: "An error occurred.", flags: 1 <<< 6}})
    end
  end

  defp run_add(ctx) do
    %{"question" => %{value: content}} = ctx.options["add"].value

    case Repo.QotdQuestion.create(ctx.guild_id, content) do
      {:ok, _question} ->
        remaining =
          ctx.guild_id
          |> Repo.QotdQuestion.get_for_guild()
          |> Enum.count(fn q -> is_nil(q.posted_at) end)

        reply(ctx, %{
          type: 4,
          data: %{
            content: "Question added. There are now #{remaining} unposted question(s) in the pool.",
            flags: 1 <<< 6
          }
        })

      {:error, err} ->
        Logger.error("Failed to add QOTD question for guild #{ctx.guild_id}: #{inspect(err)}")
        reply(ctx, %{type: 4, data: %{content: "Failed to add the question.", flags: 1 <<< 6}})
    end
  end

  defp run_remove(ctx) do
    %{"question" => %{value: id_str}} = ctx.options["remove"].value

    with {id, ""} <- Integer.parse(id_str),
         %Repo.QotdQuestion{} = question <- Repo.QotdQuestion.get(id),
         true <- question.guild_id == ctx.guild_id or :wrong_guild,
         {:ok, _} <- Repo.QotdQuestion.delete(question) do
      reply(ctx, %{
        type: 4,
        data: %{content: "Question removed.", flags: 1 <<< 6}
      })
    else
      # Integer.parse returned :error (not a number at all)
      :error ->
        reply(ctx, %{
          type: 4,
          data: %{content: "Invalid question ID. Please select a question from the list.", flags: 1 <<< 6}
        })

      # Integer.parse returned {n, leftover} — not a clean integer string
      {_n, _leftover} ->
        reply(ctx, %{
          type: 4,
          data: %{content: "Invalid question ID. Please select a question from the list.", flags: 1 <<< 6}
        })

      nil ->
        reply(ctx, %{
          type: 4,
          data: %{content: "That question doesn't exist.", flags: 1 <<< 6}
        })

      :wrong_guild ->
        reply(ctx, %{
          type: 4,
          data: %{content: "That question doesn't belong to this server.", flags: 1 <<< 6}
        })

      {:error, err} ->
        Logger.error("Failed to remove QOTD question: #{inspect(err)}")
        reply(ctx, %{type: 4, data: %{content: "Failed to remove the question.", flags: 1 <<< 6}})
    end
  end

  defp run_list(ctx) do
    questions = Repo.QotdQuestion.get_for_guild(ctx.guild_id)

    if questions == [] do
      reply(ctx, %{
        type: 4,
        data: %{content: "No questions in the pool for this server.", flags: 1 <<< 6}
      })
    else
      {posted, unposted} = Enum.split_with(questions, fn q -> not is_nil(q.posted_at) end)

      lines =
        [
          if(unposted != [],
            do: ["**Unposted (#{length(unposted)}):**"] ++ Enum.map(unposted, &format_question/1),
            else: []
          ),
          if(posted != [],
            do: ["**Posted (#{length(posted)}):**"] ++ Enum.map(posted, &format_question/1),
            else: []
          )
        ]
        |> List.flatten()

      # Truncate to fit Discord's 2000 char limit, leaving room for a note.
      content =
        lines
        |> Enum.reduce_while("", fn line, acc ->
          candidate = if acc == "", do: line, else: acc <> "\n" <> line
          if String.length(candidate) <= 1900, do: {:cont, candidate}, else: {:halt, acc}
        end)

      suffix =
        if String.length(Enum.join(lines, "\n")) > 1900,
          do: "\n*(list truncated — #{length(questions)} total)*",
          else: ""

      reply(ctx, %{
        type: 4,
        data: %{content: content <> suffix, flags: 1 <<< 6}
      })
    end
  end

  defp format_question(%Repo.QotdQuestion{id: id, content: content}) do
    preview = if String.length(content) > 80, do: String.slice(content, 0, 80) <> "…", else: content
    "• `#{id}` #{preview}"
  end
end
